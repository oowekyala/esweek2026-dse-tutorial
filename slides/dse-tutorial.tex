\begin{frame}{What they have in common}
  \begin{columns}[T]
    \column{.5\textwidth}
    \begin{itemize}
      \item \textbf{Many small compute nodes}: thousands, each slow on its own
      \item \textbf{Each owns a slice of memory}: fast to reach locally, slow or impossible across
      \item \textbf{A hierarchy of levels}: bank / rank / channel; node-local scratchpad
      \item \textbf{A narrow interface to the host}: data goes in and out through it
      \item \textbf{Programmable}: what runs where is a software decision
    \end{itemize}

\section{The problem}

\begin{frame}[fragile]{Our running example: $y = A x$}
  \begin{columns}[T]
    \column{.5\textwidth}
    \begin{align*}
      y_m &= \sum_{k} A_{m,k}\, x_k \\[.5em]
      &\text{for } m \in M,\ k \in K
    \end{align*}
    \medskip
    The two loop dimensions behave differently:
    \begin{itemize}
      \item $M$ is \textbf{parallel}: rows are independent
      \item $K$ is a \textbf{reduction}: splitting it produces partial sums that must be added up
    \end{itemize}
    \column{.46\textwidth}
    \small In Cinnamon's MLIR input:
    \begin{semiverbatim}\scriptsize
#upmem = #upmem.platform<type = v1A,
                          dpus = 64, tasklets = 24>

func.func @gemv(\%A: tensor<1024x1024xi32>,
                \%x: tensor<1024xi32>)
    attributes \{cinm.available_platforms = [#upmem]\} \{
  cinm.compute \{
    cinm.op.gemv \%A, \%x
  \}
\}
    \end{semiverbatim}
    \footnotesize One DRAM rank: 64 DPUs, up to 24 tasklets each, 64 MB MRAM and 64 KB WRAM per DPU.
  \end{columns}
\end{frame}

\begin{frame}{Tiling to the memory levels}
  \begin{columns}[T]
    \column{.5\textwidth}
    \centering
    \begin{tikzpicture}[scale=.95, every node/.style={font=\scriptsize}]
      \draw[fill=cgrid] (0,0) rectangle (3.2,3.2);
      \node[anchor=south] at (1.6,3.25) {$A$: $1024 \times 1024$};
      \foreach \i in {1,...,7} \draw[cblue, thick] (0,\i*0.4) -- (3.2,\i*0.4);
      \draw[cblue, very thick, fill=cblue!25] (0,2.0) rectangle (3.2,2.4);
      \draw[corange, very thick, fill=corange!40] (0.4,2.0) rectangle (0.9,2.4);
      % labels, stacked to the right of the matrix
      \node[anchor=west, cblue, align=left] at (3.5,2.2) {\textbf{MRAM tile}, one per worker\\ $\texttt{M.mram} \times \texttt{K.mram}$};
      \draw[cblue] (3.2,2.2) -- (3.5,2.2);
      \node[anchor=west, corange, align=left] at (3.5,1.2) {\textbf{WRAM tile}, streamed\\ $\texttt{M.wram} \times \texttt{K.wram}$};
      \draw[corange] (0.65,2.0) .. controls (1.2,1.2) .. (3.5,1.2);
      \node[anchor=west, cink2, align=left] at (3.5,0.3) {\texttt{order}: which dimension\\ is spread across DPUs first};
    \end{tikzpicture}
    \column{.46\textwidth}
    \textbf{Each choice must satisfy}
    \medskip
    \begin{itemize}\small
      \item tiles divide the extents: $\texttt{M.mram} \mid 1024$, $\texttt{K.mram} \mid 1024$
      \item inner tiles divide outer ones: $\texttt{M.wram} \mid \texttt{M.mram}$, \ldots
      \item one tile per worker:\\ $\dfrac{1024}{\texttt{M.mram}} \cdot \dfrac{1024}{\texttt{K.mram}} = \texttt{dpus} \cdot \texttt{tasklets}$
      \item what the tasklets hold fits: $\le$ MRAM, $\le$ WRAM
      \item DMA granularity: tiles are multiples of 8 bytes
    \end{itemize}
  \end{columns}
  \bigskip
  % \centering\small Every arrow in the picture is a parameter; every ``must'' is a constraint.
\end{frame}

\begin{frame}{The hardware knobs are not separate}
  \begin{columns}[T]
    \column{.5\textwidth}
    \textbf{How many DPUs? How many tasklets?}\par
    \medskip
    It is tempting to fix these first and tile afterwards, but:
    \begin{itemize}
      \item the tile count must equal the worker count: \\ $\prod (\text{extent}/\text{tile}) = \texttt{dpus}\cdot\texttt{tasklets}$
      \item more tasklets $\Rightarrow$ smaller tiles $\Rightarrow$ more launch overhead per byte
      \item more DPUs $\Rightarrow$ more transfer parallelism, more partial results to gather
    \end{itemize}
    \column{.46\textwidth}
    \begin{alertblock}{Measured on the small space}
      best latency with \texttt{tasklets} = \\
      \begin{tabular}{@{}rrrrr@{}}
        1 & 2 & 4 & \textbf{8} & 16 \\
        8.0 & 4.2 & 2.4 & \textbf{1.6} & 7.8 \\
      \end{tabular}\\[.4em]
      \footnotesize (ms). Using the most tasklets is $5\times$ slower than the best.
    \end{alertblock}
    \medskip
    \small For $4096^2$ on 32 ranks, the best configuration uses \textbf{1 tasklet} on all 2048 DPUs.
  \end{columns}
  \bigskip
  \centering So hardware and tiling parameters have to be chosen together.
\end{frame}

\section{Evaluating a configuration}

\begin{frame}{Lowering one configuration}
  \begin{columns}[T]
    \column{.5\textwidth}
    Given a configuration, Cinnamon compiles the program:
    \medskip
    \begin{itemize}\small
      \item \textbf{cinm $\to$ linalg}: the operation becomes a loop nest
      \item \textbf{linalg $\to$ cnm}: the loop nest is split over a workgroup of \texttt{dpus} $\times$ \texttt{tasklets} workers. Inputs are scattered into per-worker MRAM buffers, the result is gathered back.
      \item \textbf{WRAM tiling}: each worker's kernel loops over its MRAM tile, copying WRAM-sized pieces into the scratchpad
      \item \textbf{cnm $\to$ upmem}: transfers between host and DPUs, DPU allocation, the tasklet code
    \end{itemize}
    \column{.46\textwidth}
    \small
    \begin{tabular}{@{}ll@{}}
      \toprule
      parameter & used at \\
      \midrule
      \texttt{dpus}, \texttt{tasklets} & workgroup shape \\
      \texttt{M.mram}, \texttt{K.mram} & cnm buffer shapes \\
      \texttt{order} & scatter / gather maps \\
      \texttt{M.wram}, \texttt{K.wram} & kernel loops \\
      \bottomrule
    \end{tabular}
  \end{columns}
\end{frame}

\begin{frame}{Estimating the latency}
  \begin{columns}[T]
    \column{.48\textwidth}
    The estimate for one configuration is the sum of
    \medskip
    \begin{itemize}\small
      \item the \textbf{DPU kernel}, from a cycle-accurate DPU simulator (or a faster approximation of it)
      \item the \textbf{launch overhead}, calibrated on hardware
      \item \textbf{host work}: rearranging data before transfers, reductions on the host
      \item \textbf{transfers} to and from the DPUs
    \end{itemize}
    \medskip
    \small One estimate takes 30--60\,ms on one core with the fast simulator.
    \column{.48\textwidth}
    \centering
    \fbox{\parbox{.9\linewidth}{\centering\vspace{4em}fidelity plot:\\ simulated vs.\ measured\vspace{4em}}}
  \end{columns}
\end{frame}

\section{Searching the space}

\begin{frame}{Enumerating is not an option}
  \centering
  \begin{tabular}{@{}lrrr@{}}
    \toprule
    & feasible & one core, fast simulator & \\
    \midrule
    $1024^2$ gemv, 1 rank & 3\,234 & 3 min & \small the oracle you just used \\
    $4096^2$ gemv, 32 ranks & 26\,238 & 1.5 h & \small enumerated once, on 14 cores \\
    $2048^2$ gemm, 32 ranks & 2\,821\,332 & \textbf{47 h} & \small not enumerated \\
    \bottomrule
  \end{tabular}
  \bigskip

  Every configuration a search visits costs one simulation, so a search is judged by how good a configuration it finds for a given number of simulations.
\end{frame}

\begin{frame}{Search strategies}
  \centering\small
  \begin{tabular}{@{}lll@{}}
    \toprule
    strategy & how the next configuration is chosen & uses past results \\
    \midrule
    exhaustive & every feasible one, in order & no \\
    random & uniformly at random & no \\
    local search, simulated annealing & a neighbour of the current one & the current one \\
    genetic algorithm & recombining good ones & a population \\
    Bayesian optimisation & where a model of the cost looks promising & all of them \\
    \bottomrule
  \end{tabular}
  \bigskip

  The more a strategy learns from past results, the more it can save on simulations, and the more it costs to decide what to simulate next.
\end{frame}

\begin{frame}{Bayesian optimisation in one slide}
  \begin{columns}[T]
    \column{.55\textwidth}
    \begin{enumerate}
      \item \textbf{Initial sample}: simulate a few random configurations
      \item \textbf{Surrogate}: fit a model to the results. For any configuration it predicts a cost $\mu$ and an uncertainty $\sigma$ \\ \small usually a Gaussian process; in Cinnamon an ensemble of small neural networks
      \item \textbf{Acquisition}: score candidates by $\mu - \kappa\sigma$, so both a low predicted cost and a high uncertainty make a candidate attractive
      \item \textbf{Simulate} the best-scoring candidate, add it to the results, go to 2
    \end{enumerate}
    \column{.42\textwidth}
    \begin{tikzpicture}[scale=.85, every node/.style={font=\scriptsize}]
      \draw[->] (0,0) -- (4.6,0) node[below left] {configuration};
      \draw[->] (0,0) -- (0,3) node[left] {cost};
      \fill[cblue!15] plot[smooth] coordinates {(0.2,1.4) (1,1.1) (2,2.2) (3,1.6) (4.2,2.5)} -- plot[smooth] coordinates {(4.2,1.3) (3,1.0) (2,1.4) (1,0.7) (0.2,0.6)} -- cycle;
      \draw[cblue, thick] plot[smooth] coordinates {(0.2,1.0) (1,0.9) (2,1.8) (3,1.3) (4.2,1.9)};
      \foreach \x/\y in {0.4/1.0, 1.2/0.85, 2.3/1.9, 3.1/1.3} \fill[cink] (\x,\y) circle (2pt);
      \draw[corange, thick, dashed] (3.9,0) -- (3.9,2.5);
      \node[corange, above] at (3.9,2.5) {next: high $\sigma$};
      \draw[caqua, thick, dashed] (1.1,0) -- (1.1,0.9);
      \node[caqua, below] at (1.1,0) {low $\mu$};
      \node[cblue, right] at (2.1,2.6) {$\mu \pm \sigma$};
    \end{tikzpicture}
  \end{columns}
  \medskip
  \small The search only proposes feasible configurations, so no simulation is spent on a configuration that cannot be compiled.
\end{frame}

\begin{frame}{Take-aways}
  \begin{enumerate}
    \item The design space follows from the program and the platform: the parameters and the constraints between them can be generated rather than written by hand.
    \item Hardware parameters (DPUs, tasklets) and tiling parameters have to be chosen together. The best number of tasklets changes with the problem size.
    \item On a small instance an exhaustive run shows what the landscape looks like, and gives the optimum to compare searches against.
    \item Search results depend on the random seed: compare several runs, not one.
  \end{enumerate}
